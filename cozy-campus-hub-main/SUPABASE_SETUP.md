
# ⚡️ Supabase Setup Instructions

1.  **Create a Project:** Go to [database.new](https://database.new) and create a new project.
2.  **Get Credentials:** Go to **Project Settings > API** and copy the `URL` and `anon public` key.
3.  **Update `.env`:** Add these to your `.env` file (create one if it doesn't exist):
    ```env
    VITE_SUPABASE_URL=your-project-url
    VITE_SUPABASE_ANON_KEY=your-anon-key
    ```
4.  **Run SQL Query:** Go to the **SQL Editor** in Supabase dashboard and run the following script to set up tables and security policies:

```sql
-- Create profiles table (extends auth.users)
create table public.profiles (
  id uuid references auth.users(id) on delete cascade not null primary key,
  email text unique,
  mobile text,
  name text,
  college text,
  course text,
  role text default 'student', -- 'student' or 'admin' 
  status text default 'pending', -- 'pending', 'active', 'approved', 'rejected'
  photo_url text,
  password text, -- Added as requested
  plan text default 'Monthly',
  plan_end_date timestamptz,
  created_at timestamptz default now()
);

-- Enable RLS
alter table public.profiles enable row level security;

-- Policies for profiles
create policy "Public profiles are viewable by everyone" 
  on profiles for select using (true);

create policy "Users can update own profile" 
  on profiles for update using (auth.uid() = id);

-- Handle new user creation automatically
create or replace function public.handle_new_user() 
returns trigger as $$
begin
  insert into public.profiles (id, email, role, status)
  values (new.id, new.email, 'student', 'pending');
  return new;
end;
$$ language plpgsql security definer;

create trigger on_auth_user_created
  after insert on auth.users
  for each row execute procedure public.handle_new_user();

-- Create menu_items table
create table public.menu_items (
  id bigint generated by default as identity primary key,
  name text not null,
  category text not null, -- 'lunch' or 'dinner'
  votes int default 0,
  created_at timestamptz default now()
);

-- Enable RLS for menu
alter table public.menu_items enable row level security;

create policy "Menu items are viewable by everyone" 
  on menu_items for select using (true);

create policy "Admins can insert/update/delete menu items" 
  on menu_items for all using (
    exists (select 1 from profiles where id = auth.uid() and role = 'admin')
  );
  
-- Allow updates to votes (anyone authenticated can vote technically via RPC logic later, but for simple direct update)
create policy "Authenticated users can update votes"
  on menu_items for update using (auth.role() = 'authenticated');


-- Create votes tracking table (to prevent multiple votes)
create table public.votes (
  id bigint generated by default as identity primary key,
  user_id uuid references profiles(id) not null,
  menu_item_id bigint references menu_items(id) not null,
  category text not null, -- 'lunch' or 'dinner'
  created_at timestamptz default now(),
  unique(user_id, category) -- Enforce one vote per category per user
);

-- Enable RLS for votes
alter table public.votes enable row level security;

create policy "Users can view own votes" 
  on votes for select using (auth.uid() = user_id);

create policy "Users can insert own votes" 
  on votes for insert with check (auth.uid() = user_id);

-- Create stored procedure for voting (atomic update)
create or replace function vote_for_item(item_id bigint, user_id uuid, category_name text)
returns void as $$
begin
  -- Insert vote record (will fail if duplicate due to unique constraint)
  insert into votes (user_id, menu_item_id, category) values (user_id, item_id, category_name);
  
  -- Increment vote count on menu item
  update menu_items set votes = votes + 1 where id = item_id;
end;
$$ language plpgsql security definer;

```
